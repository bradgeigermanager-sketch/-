                const chain = document.getElementById(`iso6393-chain-${targetElementId}`);
                const resultPanel = document.getElementById(`iso6393-result-${targetElementId}`);

                // Clear all downstream elements
                while(chain.children.length > 1) chain.removeChild(chain.lastChild);
                resultPanel.classList.add('iso6393-hidden');

                if(!selectedMacro) return;

                // Add second level dropdown
                const arrow = document.createElement('div');
                arrow.className = 'iso6393-arrow';
                arrow.textContent = '→';
                chain.appendChild(arrow);

                const childWrapper = document.createElement('div');
                childWrapper.className = 'iso6393-dropdown-wrapper';
                const childSelect = document.createElement('select');
                childSelect.innerHTML = `<option value="">-- Select Individual Language --</option>`;

                    const opt = document.createElement('option');
                    opt.value = code;
                    opt.textContent = `${name} (${code})`;
                    childSelect.appendChild(opt);
                });

                childWrapper.appendChild(childSelect);
                chain.appendChild(childWrapper);

                // Render final result
                childSelect.addEventListener('change', (childE) => {
                    const selectedCode = childE.target.value;
                    if(!selectedCode) {
                        resultPanel.classList.add('iso6393-hidden');
                        return;
                    }
                    const macroInfo = _fullMacroDataset[selectedMacro];
                    resultPanel.innerHTML = `
                        <p><span class="iso6393-code-badge">${selectedMacro}</span> Macrolanguage: <strong>${macroInfo.name}</strong> (Region: ${macroInfo.scope})</p>
                        <p style="margin-top: 0.75rem;"><span class="iso6393-code-badge">${selectedCode}</span> Selected Language: <strong>${macroInfo.children[selectedCode]}</strong></p>
                        <p style="margin-top: 0.75rem; font-size: 0.9rem; color: #475569;">Entry compliant with official SIL ISO 639-3:2023 standard.</p>
                    `;
                    resultPanel.classList.remove('iso6393-hidden');
                });
            });

                getAllMacrolanguages: () => Object.keys(_fullMacroDataset).map(code => ({
                    code,
                    name: _fullMacroDataset[code].name,
                    scope: _fullMacroDataset[code].scope
                })),
                getChildrenOfMacro: (macroCode) => _fullMacroDataset[macroCode]?.children || null,
                addCustomMacro: (code, name, scope, children) => {
                    _fullMacroDataset[code] = {name, scope, children};
                    return true;
                }
            };
        }

How to Use This Module
Save the code above as iso6393-macrolang-module.js
Import it into your HTML page, initialize it with one line of code:
html
<div id="lang-navigator"></div>
<script src="./iso6393-macrolang-module.js"></script>
<script>
    // Initialize the standalone module
    const langNavigator = ISO6393MacroModule.init('lang-navigator');
    
    // Example public API usage: get all macrolanguage list
    console.log(langNavigator.getAllMacrolanguages());
</script>
